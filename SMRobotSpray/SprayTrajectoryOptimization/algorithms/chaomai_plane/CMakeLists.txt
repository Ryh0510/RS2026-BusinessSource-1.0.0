cmake_minimum_required(VERSION 3.20)
# Scope optimization to the numeric implementation, preserving viewer Debug checks.
if(MSVC)
    string(REPLACE "/RTC1" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
    string(REPLACE "/Od" "" CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG}")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/planning/uniformity_solution.json" _solution)
string(REPLACE "\n" ")RS2026\"\nR\"RS2026(" _solution "${_solution}")
set(_fingerprint "native-cpp-v1")
foreach(_source ChaomaiPlaneOptimizer.h ChaomaiPlaneOptimizer.cpp planning/rs2026_kernel.cpp planning/uniformity_solution.json)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${_source}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${_source}" _digest)
    string(APPEND _fingerprint "${_digest}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../../../../SimWorkbench/SMRobotWorkbenchPaintingAnalysis/TrajectoryOptimizationWorkbench/TrajectoryOptimizationService.cpp" _adapter_digest)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../../SimWorkbench/SMRobotWorkbenchPaintingAnalysis/TrajectoryOptimizationWorkbench/TrajectoryOptimizationService.cpp")
string(SHA256 _fingerprint "${_fingerprint}${_adapter_digest}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/ChaomaiPlaneData.h" CONTENT
    "#pragma once\nstatic const char* savedSolution = R\"RS2026(${_solution})RS2026\";\nstatic const char* nativeImplementationSignature = \"${_fingerprint}\";\n")
add_library(ChaomaiPlaneOptimizer OBJECT ChaomaiPlaneOptimizer.cpp planning/rs2026_kernel.cpp)
target_compile_features(ChaomaiPlaneOptimizer PUBLIC cxx_std_17)
target_include_directories(ChaomaiPlaneOptimizer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ChaomaiPlaneOptimizer PUBLIC nlohmann_json::nlohmann_json PRIVATE Eigen3::Eigen)
set_target_properties(ChaomaiPlaneOptimizer PROPERTIES FOLDER SMRobotSpray AUTOMOC OFF)
if(MSVC)
    target_compile_options(ChaomaiPlaneOptimizer PRIVATE /O2 /Ob2 /utf-8)
endif()
